public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        int choix = 0;

        Client client = new Client(1, "Balla", "bgc@gmail.com");

        Produit produit1 = new Produit(1, "Souris", 39.99f, 5);
        Produit produit2 = new Produit(2, "Clavier", 59.99f, 3);
        Produit produit3 = new Produit(3, "Casque", 29.99f, 7);

        Magasin magasin = new Magasin();
        magasin.ajouterProduit(produit1);
        magasin.ajouterProduit(produit2);
        magasin.ajouterProduit(produit3);

        Panier panier = new Panier();

        while (choix != 5) {
            System.out.println("\n--- Menu Magasin ---");
            System.out.println("1. Afficher les produits disponibles");
            System.out.println("2. Ajouter un produit au panier");
            System.out.println("3. Afficher le panier");
            System.out.println("4. Passer la commande");
            System.out.println("5. Quitter");
            System.out.print("Choisissez une option : ");

            choix = sc.nextInt();

            switch (choix) {
                case 1:
                    magasin.afficherProduitsDisponibles();
                    break;

                case 2:
                    System.out.print("Entrez le nom du produit à ajouter : ");
                    String nom = sc.next();
                    Produit produitTrouve = magasin.trouverProduitParNom(nom);
                    if (produitTrouve != null) {
                        panier.ajouterProduit(produitTrouve);
                        System.out.println("Produit ajouté au panier !");
                    } else {
                        System.out.println("Produit introuvable.");
                    }
                    break;

                case 3:
                    panier.afficherPanier();
                    System.out.println("Total : " + panier.calculerTotal() + " €");
                    break;

                case 4:
                    if (panier.getProduits().isEmpty()) {
                        System.out.println("Votre panier est vide.");
                    } else {
                        Commande commande = new Commande(1, client, panier.getProduits(), panier.calculerTotal());
                        commande.afficherDetailsCommande();
                    }
                    break;

                case 5:
                    System.out.println("Au revoir !");
                    break;

                default:
                    System.out.println("Option invalide.");
                    break;
            }
        }

        sc.close();
    }
}
